# ---------- Web 框架 ----------
# fastapi：低于 0.100 不支持 pydantic v2，会被拖回 v1（用法完全不同）
fastapi>=0.110
httpx>=0.27
# python-multipart：FastAPI 处理 multipart 表单（UploadFile 文件上传）必需；
#   缺了启动时直接报 "Form data requires python-multipart to be installed"
python-multipart>=0.0.9
# uvicorn[standard]：[standard] 才带 --reload(watchfiles)/httptools；
#   Windows 上不含 uvloop（uvloop 仅 *nix），别指望它在 Windows 生效
uvicorn[standard]>=0.29

# ---------- 数据库 / ORM（异步）----------
# sqlalchemy：必须 2.0+。1.4 的 async 是实验特性且 API 不同，混用踩坑
sqlalchemy[asyncio]>=2.0
# aiomysql：异步 MySQL 驱动。维护偏慢，若在 Python 3.13 上安装/运行异常，
#   改用 asyncmy（连接串相应改为 mysql+asyncmy://）
aiomysql>=0.2.0
# cryptography：MySQL 8 默认 caching_sha2_password 认证必需；
#   缺了会报 "cryptography is required for sha256_password" 之类
cryptography>=42
# greenlet：SQLAlchemy async 运行时依赖。C 扩展，Python 小版本升级常缺预编译 wheel，
#   版本过旧会在 3.13 上装不上；不够新就升级它
greenlet>=3.1
# alembic：必须 1.11+。1.10 及以下不支持 SQLAlchemy 2.0（迁移会直接报错）
alembic>=1.13

# ---------- 任务队列 / 缓存（2026-10-16 RAG 索引任务链路，运行必需）----------
# redis：用官方 redis.asyncio（异步）。**别用已废弃的 aioredis**（独立包，勿装）。
#   ⚠️ 用途已变：原先标注「M0-M2 不要求 Redis 服务」，现在 Redis 是
#   **arq 任务队列的唯一通道**，缺了它 worker 完全跑不起来（设计见
#   docs/RAG索引任务-设计.md）。Redis 服务需常驻。
redis>=5.0
# arq：异步任务队列。**选它而不是 RQ 的唯一理由是「异步原生」**——
#   arq 的 worker 任务函数是 async def，与本项目全异步栈（SQLAlchemy 2.x 异步 +
#   aiomysql/asyncpg + httpx.AsyncClient）同构；RQ 的 worker 是同步的，要用
#   asyncio.run()包一层、每次建销事件循环、AsyncClient 无法跨任务复用。
#   用法：arq app.worker.WorkerSettings（见 docs/启动命令.md §6）
#   ⚠️ arq 作者已声明该项目 **maintenance only**（仅修 bug，无新功能）。功能上够用；
#      将来若需要任务优先级/结果面板，得换库——但迁移成本低（任务函数是普通
#      async def，只改 WorkerSettings 与投递那几行）。
arq>=0.28
# Pillow：本轮（第 2 轮）就需要 —— 人机验证已定为自研图形码，验证码图片由 Pillow 生成。
#   改用腾讯验证码/极验这类外部服务才不需要它。
Pillow>=10.3

# ---------- 配置 / 校验 / 安全 ----------
# pydantic：必须 v2。v1 的 BaseSettings/models 配置写法与 v2 完全不同
pydantic>=2.7
# pydantic-settings：v2 起 BaseSettings 从 pydantic 拆到独立包，
#   必须 from pydantic_settings import BaseSettings（从 pydantic 引会报错）
pydantic-settings>=2.2
# email-validator：Pydantic EmailStr 的运行时依赖，不加会 ImportError
email-validator>=2.1
# bcrypt：直接用，别配 passlib。passlib 1.7.4 读不到 bcrypt 4.x 的 __about__，会告警甚至报错
bcrypt>=4.1

# ---------- PostgreSQL（PG 16 + pgvector 作向量库）----------
# asyncpg：PG 的异步驱动，PG_URL 用 postgresql+asyncpg://
#   ⚠️ 它的连接串 scheme 是 **postgresql+asyncpg**，不是 mysql+aiomysql 那种写法
asyncpg>=0.29
langgraph>=1.2.4
